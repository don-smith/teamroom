# M4.5 — AI-Native Harness Implementation Plan

> **For agentic workers:** REQUIRED SUB-SKILL: Use `superpowers:subagent-driven-development` (recommended) or `superpowers:executing-plans` to implement this plan task-by-task. Steps use checkbox (`- [ ]`) syntax for tracking.

**Goal:** Replace the legacy CLI orchestration path (Orchestrator + tail + FileNotifyBus + DB-direct write + `/cli/send` on mail-MCP) with a channel-architecture surface (CLI as the first concrete channel; daemon-owned WebSocket endpoints for chat and mail) while extracting `startDaemon`'s 322-line procedural wire-up into eight named modules under `src/modules/` and retrofitting the mail-MCP server onto `@modelcontextprotocol/sdk`.

**Architecture:** Eight `compose<Module>(deps)` factory functions live under `src/modules/`. `startDaemon` becomes a thin top-level composer that calls them in order; `composeIntegrationHarness` calls the same composers with test-friendly substitutions. Composition parity is delivered by construction (same call graph, different concrete deps). Channels live as a registry + a daemon-owned chat WebSocket; the CLI command is a separate process that connects as a WebSocket client, registers as `channel:cli:<user>`, and receives bus events addressed to its channel.

**Tech Stack:** TypeScript strict + ESM, vitest, better-sqlite3, `@modelcontextprotocol/sdk`, `ws` (WebSocket library), node:http, Docker for agent containers.

**Spec deviation flagged for review:** the spec describes the address-scheme migration as TS code in the migrator. The plan uses plain SQL with SQLite JSON1 (`json_each` + `json_group_array`) — works, keeps the migrator unchanged, no extension needed. Single forward-only `004_channel_address_scheme.sql` file. If you'd prefer the migrator extension, swap Phase 3 for a TS-migration variant.

```typescript
import { mkdirSync, appendFileSync } from "node:fs";
import { join } from "node:path";
import { SystemClock, type Clock } from "../../clock/clock.js";
import { JsonLinesLogger, type Logger } from "../../logger/logger.js";

export interface Observability {
  clock: Clock;
  rootLogger: Logger;
  logFilePath: string;
}

export interface ObservabilityDeps {
  dataDir: string;
}

export function composeObservability(deps: ObservabilityDeps): Observability {
  mkdirSync(join(deps.dataDir, "logs"), { recursive: true });
  const clock = new SystemClock();
  const dateStr = clock.nowIso().slice(0, 10);
  const logFilePath = join(deps.dataDir, "logs", `teamroom-${dateStr}.jsonl`);
  const logSink = (line: string) => {
    process.stdout.write(line + "\n");
    appendFileSync(logFilePath, line + "\n");
  };
  const rootLogger = new JsonLinesLogger(clock, logSink, { component: "daemon" });
  return { clock, rootLogger, logFilePath };
}
```

```typescript
import { mkdirSync } from "node:fs";
import { join } from "node:path";
import { createDatabase, type Db } from "../../db/database.js";
import { runMigrations } from "../../db/migrator.js";
import { AgentRepo } from "../../db/agent-repo.js";
import { MessageRepo } from "../../db/message-repo.js";
import { TurnRepo } from "../../db/turn-repo.js";

export interface Storage {
  db: Db;
  agentRepo: AgentRepo;
  messageRepo: MessageRepo;
  turnRepo: TurnRepo;
  close(): void;
}

export interface StorageDeps {
  /** Either a directory path (file DB) or `:memory:` for in-process tests. */
  dataDir: string;
}

export function composeStorage(deps: StorageDeps): Storage {
  let dbPath: string;
  if (deps.dataDir === ":memory:") {
    dbPath = ":memory:";
  } else {
    mkdirSync(deps.dataDir, { recursive: true });
    dbPath = join(deps.dataDir, "teamroom.db");
  }
  const db = createDatabase(dbPath);
  runMigrations(db);
  return {
    db,
    agentRepo: new AgentRepo(db),
    messageRepo: new MessageRepo(db),
    turnRepo: new TurnRepo(db),
    close: () => db.close(),
  };
}
```

```typescript
import { mkdirSync, writeFileSync, readFileSync, unlinkSync, existsSync } from "node:fs";
import { join } from "node:path";

export function portsDir(dataDir: string): string {
  return join(dataDir, "ports");
}

export function writePortFile(dataDir: string, name: string, port: number): void {
  mkdirSync(portsDir(dataDir), { recursive: true });
  writeFileSync(join(portsDir(dataDir), `${name}.port`), String(port));
}

export function readPortFile(dataDir: string, name: string): number | null {
  const path = join(portsDir(dataDir), `${name}.port`);
  if (!existsSync(path)) return null;
  const raw = readFileSync(path, "utf8").trim();
  const port = Number.parseInt(raw, 10);
  return Number.isFinite(port) && port > 0 ? port : null;
}

export function deletePortFile(dataDir: string, name: string): void {
  try {
    unlinkSync(join(portsDir(dataDir), `${name}.port`));
  } catch {
    /* best-effort */
  }
}
```

`src/mail-dispatcher/address.ts`:

```typescript
/**
 * M4.5 addresses are hierarchical:
 *   agent:<name>                       — an agent's inbox
 *   list:<name>                        — a mailing list (resolved at dispatch)
 *   channel:<channel>:<external-id>    — a channel-routed external recipient
 *
 * Lenient parse / strict deliver: any well-formed address parses; the
 * dispatcher rejects channel addresses whose `channel` isn't in the
 * channel registry at delivery time. Mirrors how email itself works.
 */

export type Address =
  | { kind: "agent"; name: string }
  | { kind: "list"; name: string }
  | { kind: "channel"; channel: string; externalId: string };

const TOP_LEVEL_PREFIXES = new Set(["agent", "list", "channel"]);

export function parseAddress(s: string): Address {
  const idx = s.indexOf(":");
  if (idx <= 0 || idx === s.length - 1) throw new Error(`invalid address: ${s}`);
  const prefix = s.slice(0, idx);
  const rest = s.slice(idx + 1);
  if (!TOP_LEVEL_PREFIXES.has(prefix)) {
    throw new Error(`unknown address prefix: ${prefix}`);
  }
  if (prefix === "agent") return { kind: "agent", name: rest };
  if (prefix === "list") return { kind: "list", name: rest };
  // channel:<channel>:<external-id>
  const colonIdx = rest.indexOf(":");
  if (colonIdx <= 0 || colonIdx === rest.length - 1) {
    throw new Error(`invalid channel address: ${s}`);
  }
  return {
    kind: "channel",
    channel: rest.slice(0, colonIdx),
    externalId: rest.slice(colonIdx + 1),
  };
}

export function formatAddress(a: Address): string {
  switch (a.kind) {
    case "agent": return `agent:${a.name}`;
    case "list": return `list:${a.name}`;
    case "channel": return `channel:${a.channel}:${a.externalId}`;
  }
}
```

```sql
-- Migration 004: channel address scheme
--
-- Rewrites legacy `cli:<user>` addresses to the M4.5 hierarchical form
-- `channel:cli:<user>`. Three columns affected: `from` (single string),
-- `to` (JSON array), `cc` (JSON array, nullable).
--
-- Idempotent: a re-run after a successful first pass rewrites zero rows
-- because the `LIKE 'cli:%'` predicates no longer match.

-- Update `from` (single string column).
UPDATE messages
SET "from" = 'channel:' || "from"
WHERE "from" LIKE 'cli:%';

-- Update `to` (JSON array).
UPDATE messages
SET "to" = (
  SELECT json_group_array(
    CASE WHEN value LIKE 'cli:%' THEN 'channel:' || value ELSE value END
  )
  FROM json_each(messages."to")
)
WHERE EXISTS (
  SELECT 1 FROM json_each(messages."to") WHERE value LIKE 'cli:%'
);

-- Update `cc` (JSON array, nullable).
UPDATE messages
SET cc = (
  SELECT json_group_array(
    CASE WHEN value LIKE 'cli:%' THEN 'channel:' || value ELSE value END
  )
  FROM json_each(messages.cc)
)
WHERE cc IS NOT NULL
  AND EXISTS (
    SELECT 1 FROM json_each(messages.cc) WHERE value LIKE 'cli:%'
```

```typescript
import type { DockerRunSpec, Mount } from "../container/docker-command.js";
import type { AgentConfig } from "../agent-config/agent-config.js";
import type { TokenRegistry } from "../model-proxy/token-registry.js";

export interface BuildSpecArgs {
  agentId: string;
  mcpToken: string;
  agentConfigs: Map<string, AgentConfig>;
  tokenRegistry: TokenRegistry;
  modelBaseUrl: string;
  mailPort: number;
  codebasePath: string | null;
  genId: () => string;
}

/**
 * Build the DockerRunSpec for one agent session. Lifted from the inline
 * closure in startDaemon so its dep contract is explicit and it can be
 * tested in isolation.
 */
export function buildSpec(args: BuildSpecArgs): DockerRunSpec {
  const cfg = args.agentConfigs.get(args.agentId);
  if (!cfg) throw new Error(`no agent config for '${args.agentId}'`);
  const proxyToken = args.tokenRegistry.mint({
    agentId: args.agentId,
    turnId: `session-${args.mcpToken.slice(0, 8)}`,
    traceId: args.genId(),
    provider: cfg.provider,
    model: cfg.model,
  });
  const mounts: Mount[] = [
    { hostPath: cfg.agentsMdPath, containerPath: "/AGENTS.md", readOnly: false },
    { hostPath: cfg.storePath, containerPath: "/store", readOnly: false },
  ];
  if (args.codebasePath) {
    mounts.push({ hostPath: args.codebasePath, containerPath: "/codebase", readOnly: false });
  }
  return {
    imageTag: cfg.imageTag,
    name: `teamroom-${args.agentId}-${args.mcpToken.slice(0, 8)}`,
    mounts,
    env: {
      TEAMROOM_TRACE_ID: args.genId(),
      TEAMROOM_AGENT_NAME: args.agentId,
      TEAMROOM_PROVIDER: cfg.provider,
      TEAMROOM_MODEL: cfg.model,
      TEAMROOM_MODEL_BASE_URL: args.modelBaseUrl,
      TEAMROOM_MODEL_API_TOKEN: proxyToken,
      ANTHROPIC_BASE_URL: args.modelBaseUrl,
      ANTHROPIC_AUTH_TOKEN: proxyToken,
      TEAMROOM_ALLOWED_TOOLS: cfg.allowedTools.join(","),
      TEAMROOM_MCP_URL: `http://host.docker.internal:${args.mailPort}/mcp`,
      TEAMROOM_MCP_TOKEN: args.mcpToken,
    },
    labels: {
      "teamroom-agent": "true",
      "teamroom-agent-id": args.agentId,
    },
  };
}
```

```typescript
  logger: Logger;

  agentConfigs: Map<string, AgentConfig>;
  lists: ListDefinition[];

export function composeMcpMail(deps: McpMailDeps): McpMail {
  const mailingListRegistry = new MailingListRegistry(
    deps.lists,
    [...deps.agentConfigs.values()],
  );
  deps.logger.info("mailing-list registry built", {
    lists: deps.lists.length,

    listSubscribers: (n) => mailingListRegistry.subscribers(n),
    maxMailsPerThread: deps.maxMailsPerThread,

  // wants to inspect it.

  const _triggerListener = new MailingListTriggerListener({

    messageRepo: deps.storage.messageRepo,
    dispatcher,
    bus: deps.messaging.eventBus,
    clock: deps.clock,
  });
```

```typescript
export class ChannelRegistry {
  private readonly channels = new Set<string>();

  register(name: string): void {

    }
    this.channels.add(name);
  }

  has(name: string): boolean {

  }

  listChannels(): string[] {
    return [...this.channels];
  }
```

```typescript
export type ClientMessage =
  | { type: "register"; channel: string; externalId: string }
  | { type: "publish"; to: string[]; body: string; threadId: string | null; cc: string[] };

export type ServerMessage =
  | { type: "mail"; id: string; from: string; to: string[]; cc: string[]; body: string; threadId: string; ts: string }
  | { type: "event"; event: string; payload: unknown }
  | { type: "error"; code: string; message: string };

export function parseClientMessage(raw: string): ClientMessage {
  const parsed = JSON.parse(raw);
  if (typeof parsed !== "object" || parsed === null || typeof parsed.type !== "string") {
    throw new Error("invalid client message: missing type");
  }
  if (parsed.type === "register") {
    if (typeof parsed.channel !== "string" || typeof parsed.externalId !== "string") {
      throw new Error("register: channel and externalId are required strings");
    }
    return { type: "register", channel: parsed.channel, externalId: parsed.externalId };
  }
  if (parsed.type === "publish") {
    if (!Array.isArray(parsed.to) || typeof parsed.body !== "string") {
      throw new Error("publish: to (array) and body (string) are required");
    }
    return {
      type: "publish",
      to: parsed.to,
      body: parsed.body,
      threadId: parsed.threadId ?? null,
      cc: Array.isArray(parsed.cc) ? parsed.cc : [],
    };
  }
  throw new Error(`unknown client message type: ${parsed.type}`);
}
```

```typescript
  private readonly connections = new Set<RegisteredConnection>();

  constructor(private readonly opts: ChatServerOpts) {}

  get port(): number { return this.listenPort; }

  async start(): Promise<void> {
    this.http = createServer();
    this.wss = new WebSocketServer({ server: this.http });
    this.wss.on("connection", (ws) => this.handleConnection(ws));

    // Subscribe to mail.delivered for channel-addressed recipients.
    this.opts.messaging.eventBus.subscribe<{ to: string; messageId: string }>(
      "mail.delivered",
      (payload) => this.routeDelivery(payload),
    );

    await new Promise<void>((resolve) => this.http!.listen(this.opts.port, () => resolve()));
    const addr = this.http!.address();
    this.listenPort = typeof addr === "object" && addr ? addr.port : 0;
    this.opts.logger.info("chat server started", { port: this.listenPort });
  }

  async stop(): Promise<void> {
    if (this.wss) await new Promise<void>((r) => this.wss!.close(() => r()));
    if (this.http) await new Promise<void>((r) => this.http!.close(() => r()));
    this.wss = null; this.http = null;
  }

  private handleConnection(ws: WebSocket): void {
    let registered: RegisteredConnection | null = null;
    ws.on("message", (raw) => {
      let msg;
      try { msg = parseClientMessage(raw.toString()); }
      catch (err) {
        this.send(ws, { type: "error", code: "invalid_message", message: String(err) });
        return;
      }
      if (msg.type === "register") {
        if (!this.opts.registry.has(msg.channel)) {
          this.send(ws, { type: "error", code: "unknown_channel", message: `channel '${msg.channel}' not registered` });
          return;
        }
        if (registered) this.connections.delete(registered);
        registered = { channel: msg.channel, externalId: msg.externalId, ws };
        this.connections.add(registered);
        return;
      }
      if (msg.type === "publish") {
        if (!registered) {
          this.send(ws, { type: "error", code: "not_registered", message: "register before publish" });
          return;
        }
        try {
          this.opts.mcpMail.dispatcher.sendMail({
            from: `channel:${registered.channel}:${registered.externalId}`,
            to: msg.to,
            body: msg.body,
            threadId: msg.threadId,
            cc: msg.cc.length > 0 ? msg.cc : undefined,
          });
        } catch (err) {
          this.send(ws, { type: "error", code: "dispatch_failed", message: String(err) });
        }
      }
    });
    ws.on("close", () => { if (registered) this.connections.delete(registered); });
  }

  private routeDelivery(payload: { to: string; messageId: string }): void {
    let addr;
    try { addr = parseAddress(payload.to); } catch { return; }
    if (addr.kind !== "channel") return;
    const target = [...this.connections].find(
      (c) => c.channel === addr.channel && c.externalId === addr.externalId,
    );
    if (!target) return;
    // Look up the message body to forward.
    // (ChatServer has a dep on mcpMail; can read storage via mcpMail.dispatcher's
    //  messageRepo, or take a direct storage dep. Plan choice: take storage dep.)
    // ... fetch full mail row, send `{ type: "mail", ... }`
  }

  private send(ws: WebSocket, msg: ServerMessage): void {
    ws.send(JSON.stringify(msg));
  }
}
```

```typescript
export type ClientMessage =
  | { type: "query"; id: string; query: "inbox"; params: { agent: string } }
  | { type: "query"; id: string; query: "thread"; params: { id: string } };

export type ServerMessage =
  | { type: "result"; id: string; rows: unknown[] }
  | { type: "error"; id: string; code: string; message: string };

const QUERIES = new Set(["inbox", "thread"]);

export function parseClientMessage(raw: string): ClientMessage {
  const p = JSON.parse(raw);
  if (p?.type !== "query" || typeof p.id !== "string" || typeof p.query !== "string") {
    throw new Error("invalid client message");
  }
  if (!QUERIES.has(p.query)) throw new Error(`unknown query: ${p.query}`);
  if (p.query === "inbox") {
    if (typeof p.params?.agent !== "string") throw new Error("inbox: params.agent required");
    return { type: "query", id: p.id, query: "inbox", params: { agent: p.params.agent } };
  }
  if (p.query === "thread") {
    if (typeof p.params?.id !== "string") throw new Error("thread: params.id required");
    return { type: "query", id: p.id, query: "thread", params: { id: p.params.id } };
  }
  throw new Error("unreachable");
}
```

```typescript
import { WebSocket } from "ws";
import { createInterface, type Interface as Readline } from "node:readline";
import type { Readable, Writable } from "node:stream";

export interface RunChatArgs {
  port: number;
  agent: string;
  user: string;
  initialBody?: string;
  stdin: Readable;
  stdout: Writable;
}

export async function runChat(args: RunChatArgs): Promise<void> {
  const ws = new WebSocket(`ws://127.0.0.1:${args.port}/`);
  await new Promise<void>((ok, fail) => {
    ws.once("open", () => ok());
    ws.once("error", (err) => fail(err));
  });
  ws.send(JSON.stringify({ type: "register", channel: "cli", externalId: args.user }));

  ws.on("message", (raw) => {
    const msg = JSON.parse(raw.toString());
    if (msg.type === "mail") {
      args.stdout.write(`${msg.from}: ${msg.body}\n`);
    } else if (msg.type === "error") {
      args.stdout.write(`error: ${msg.code} — ${msg.message}\n`);
    }
  });

  let threadId: string | null = null;

  const publish = (body: string): void => {
    ws.send(JSON.stringify({
      type: "publish",
      to: [`agent:${args.agent}`],
      body,
      threadId,
      cc: [],
    }));
    // Note: the assigned threadId isn't echoed back to the client today;
    // for v1 we keep null and let the dispatcher allocate per-message,
    // OR we extend the protocol to surface it. For simplicity and to keep
    // this milestone scoped, accept "thread per message" behavior unless
    // the spec is later updated.
  };

  if (args.initialBody) publish(args.initialBody);

  const rl: Readline = createInterface({ input: args.stdin });
  for await (const line of rl) {
    const trimmed = line.trim();
    if (trimmed === "/quit" || trimmed === "/exit") break;
    if (trimmed.length === 0) continue;
    publish(trimmed);
  }
  ws.close();
}
```
